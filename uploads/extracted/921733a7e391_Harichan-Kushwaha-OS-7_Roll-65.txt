Implement Reader-writer using Thread and Semaphore only Program:

#include <stdio.h> #include <pthread.h> #include <unistd.h>

#define SIZE 5

#define ITEMS 10

int buffer[SIZE];

int in = 0, out = 0, count = 0;

pthread_mutex_t mutex;

void *producer(void *arg)

{

for (int item = 1; item <= ITEMS; item++)

{

while (1)

{

pthread_mutex_lock(&mutex);

if (count < SIZE)

{

buffer[in] = item;

printf("Producer produced: %d\n", item);

in = (in + 1) % SIZE;

count++;

pthread_mutex_unlock(&mutex); break;

}

pthread_mutex_unlock(&mutex); usleep(10000);

}

}

return NULL;

}

void *consumer(void *arg)

{

for (int i = 1; i <= ITEMS; i++)

{

while (1)

{

pthread_mutex_lock(&mutex);

if (count > 0)

{

int item = buffer[out];

printf("Consumer consumed: %d\n", item);

out = (out + 1) % SIZE; count--;

pthread_mutex_unlock(&mutex); break;

}

pthread_mutex_unlock(&mutex); usleep(10000);

}

}

return NULL;

}

int main()

{

pthread_t p, c;

pthread_mutex_init(&mutex, NULL);

pthread_create(&p, NULL, producer, NULL); pthread_create(&c, NULL, consumer, NULL);

pthread_join(p, NULL); pthread_join(c, NULL);

pthread_mutex_destroy(&mutex);

return 0;
